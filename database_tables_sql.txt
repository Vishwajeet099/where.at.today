-- Enable UUID generation
CREATE EXTENSION IF NOT EXISTS pgcrypto;

--------------------------------------------------
-- 1. CITIES
--------------------------------------------------
CREATE TABLE cities (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100) NOT NULL,
  slug VARCHAR(100) UNIQUE NOT NULL,
  country VARCHAR(100) DEFAULT 'India',
  timezone VARCHAR(100) DEFAULT 'Asia/Kolkata',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 2. VENUES
--------------------------------------------------
CREATE TABLE venues (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  city_id UUID NOT NULL REFERENCES cities(id) ON DELETE CASCADE,
  name VARCHAR(255) NOT NULL,
  slug VARCHAR(255) UNIQUE NOT NULL,
  address TEXT,
  latitude NUMERIC,
  longitude NUMERIC,
  capacity INTEGER,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 3. CATEGORIES
--------------------------------------------------
CREATE TABLE categories (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100) NOT NULL,
  slug VARCHAR(100) UNIQUE NOT NULL,
  icon_key VARCHAR(100),
  color_hex VARCHAR(10),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 4. EVENTS (MAIN ENTITY)
--------------------------------------------------
CREATE TABLE events (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  city_id UUID NOT NULL REFERENCES cities(id) ON DELETE CASCADE,
  title VARCHAR(255) NOT NULL,
  slug VARCHAR(255) UNIQUE NOT NULL,
  description TEXT,
  status VARCHAR(20) DEFAULT 'draft',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 5. EVENT INSTANCES (MOST IMPORTANT)
--------------------------------------------------
CREATE TABLE event_instances (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  event_id UUID NOT NULL REFERENCES events(id) ON DELETE CASCADE,
  venue_id UUID NOT NULL REFERENCES venues(id) ON DELETE CASCADE,
  start_at TIMESTAMP NOT NULL,
  end_at TIMESTAMP,
  status VARCHAR(20) DEFAULT 'scheduled',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 6. EVENT CATEGORIES (MANY-TO-MANY)
--------------------------------------------------
CREATE TABLE event_categories (
  event_id UUID REFERENCES events(id) ON DELETE CASCADE,
  category_id UUID REFERENCES categories(id) ON DELETE CASCADE,
  is_primary BOOLEAN DEFAULT FALSE,
  PRIMARY KEY (event_id, category_id)
);

--------------------------------------------------
-- 7. TAGS
--------------------------------------------------
CREATE TABLE tags (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100) NOT NULL,
  slug VARCHAR(100) UNIQUE NOT NULL
);

--------------------------------------------------
-- 8. EVENT TAGS
--------------------------------------------------
CREATE TABLE event_tags (
  event_id UUID REFERENCES events(id) ON DELETE CASCADE,
  tag_id UUID REFERENCES tags(id) ON DELETE CASCADE,
  PRIMARY KEY (event_id, tag_id)
);

--------------------------------------------------
-- 9. EVENT MEDIA
--------------------------------------------------
CREATE TABLE event_media (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  event_id UUID REFERENCES events(id) ON DELETE CASCADE,
  media_url TEXT NOT NULL,
  is_hero BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 10. ORGANIZERS
--------------------------------------------------
CREATE TABLE organizers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  slug VARCHAR(255) UNIQUE NOT NULL,
  description TEXT,
  website TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 11. EVENT ORGANIZERS (MANY-TO-MANY)
--------------------------------------------------
CREATE TABLE event_organizers (
  event_id UUID REFERENCES events(id) ON DELETE CASCADE,
  organizer_id UUID REFERENCES organizers(id) ON DELETE CASCADE,
  PRIMARY KEY (event_id, organizer_id)
);

--------------------------------------------------
-- 12. PERFORMERS / ARTISTS
--------------------------------------------------
CREATE TABLE performers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  slug VARCHAR(255) UNIQUE NOT NULL,
  bio TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 13. EVENT PERFORMERS
--------------------------------------------------
CREATE TABLE event_performers (
  event_id UUID REFERENCES events(id) ON DELETE CASCADE,
  performer_id UUID REFERENCES performers(id) ON DELETE CASCADE,
  PRIMARY KEY (event_id, performer_id)
);

--------------------------------------------------
-- 14. TICKET SELLERS
--------------------------------------------------
CREATE TABLE ticket_sellers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  website TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- 15. EVENT TICKETS
--------------------------------------------------
CREATE TABLE event_tickets (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  event_instance_id UUID REFERENCES event_instances(id) ON DELETE CASCADE,
  seller_id UUID REFERENCES ticket_sellers(id) ON DELETE CASCADE,
  ticket_url TEXT NOT NULL,
  price NUMERIC,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

--------------------------------------------------
-- INDEXES (IMPORTANT FOR PERFORMANCE)
--------------------------------------------------
CREATE INDEX idx_events_city ON events(city_id);
CREATE INDEX idx_event_instances_event ON event_instances(event_id);
CREATE INDEX idx_event_instances_venue ON event_instances(venue_id);
CREATE INDEX idx_event_instances_start ON event_instances(start_at);
CREATE INDEX idx_venues_city ON venues(city_id);
